# I1023: on macOS the selecta target is a bundle, so a double-click on Selecta.app opens the desktop UI with no Terminal
# window. $<TARGET_FILE:selecta> then names Selecta.app/Contents/MacOS/Selecta, which is how every test finds the binary.
# The icon (apps/selecta/icon/selecta.svg) is rendered into Selecta.icns at build time by icon/make_icns.sh.
if(APPLE)
  set(SELECTA_ICNS "${CMAKE_CURRENT_BINARY_DIR}/Selecta.icns")
  add_custom_command(OUTPUT "${SELECTA_ICNS}"
    COMMAND /bin/sh "${CMAKE_CURRENT_SOURCE_DIR}/icon/make_icns.sh" "${CMAKE_CURRENT_SOURCE_DIR}/icon/selecta.svg" "${SELECTA_ICNS}"
    DEPENDS icon/selecta.svg icon/make_icns.sh VERBATIM)
  target_sources(selecta PRIVATE "${SELECTA_ICNS}")
  set_source_files_properties("${SELECTA_ICNS}" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set_target_properties(selecta PROPERTIES
    MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in"
    MACOSX_BUNDLE_BUNDLE_NAME Selecta
    MACOSX_BUNDLE_GUI_IDENTIFIER se.spotup.selecta
    MACOSX_BUNDLE_ICON_FILE Selecta.icns
    OUTPUT_NAME Selecta)
  # I1162e: before I1023 the target wrote a plain executable <bin>/apps/selecta/selecta. Making it a bundle left that file
  # behind in every build tree, and a run of the old path started a 3 Nov binary that refused the v18 library silently.
  # The path now holds a launcher for the bundle binary, rewritten at every generate, so the documented and habitual
  # command `build/<tree>/apps/selecta/selecta` always runs the current build (text UI from a terminal, I1023 default).
  # A shell launcher, not a symlink: the bundle binary finds Contents/PlugIns from its own path.
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/selecta"
    CONTENT "#!/bin/sh\n# Generated by apps/selecta/CMakeLists.txt (I1162e): runs this build tree's Selecta.app binary.\nexec \"$(dirname \"$0\")/Selecta.app/Contents/MacOS/Selecta\" \"$@\"\n"
    FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
endif()

# I427f: the bundle carries the built-in modules in Contents/PlugIns, where the loader looks first after the env
# override. A target of its own, built with ALL, so a rebuilt module never leaves a stale copy behind.
if(APPLE AND SELECTA_PLUGIN_LINK STREQUAL "MODULE")
  get_property(_sel_bundle_mods GLOBAL PROPERTY SEL_BUILTIN_MODULES)
  # I1075: every standalone module (sel_add_plugin ... MODULE) ships too; a release build keeps only the MIT ones (I427d D5).
  get_property(_sel_standalone_mods GLOBAL PROPERTY SEL_STANDALONE_MODULES)
  foreach(_m IN LISTS _sel_standalone_mods)
    get_target_property(_mit ${_m} SEL_MODULE_MIT)
    if(NOT SELECTA_RELEASE_MIT_ONLY OR _mit)
      list(APPEND _sel_bundle_mods ${_m})
    else()
      set_property(GLOBAL APPEND PROPERTY SEL_LEFT_OUT_MODULES ${_m}) # tests/dui/bundle_plugins.cmake: these must not be in the bundle
    endif()
  endforeach()
  set_property(GLOBAL PROPERTY SEL_BUNDLED_MODULES ${_sel_bundle_mods}) # tests/dui/bundle_plugins.cmake counts the bundle from this
  set(_sel_plugins_dir "$<TARGET_BUNDLE_CONTENT_DIR:selecta>/PlugIns")
  add_custom_target(selecta_bundle_modules ALL
    COMMAND ${CMAKE_COMMAND} -E make_directory "${_sel_plugins_dir}"
    VERBATIM)
  # D6: the shared selecta_core travels with the bundle in Contents/Frameworks; the executable finds it through @rpath.
  set(_sel_frameworks_dir "$<TARGET_BUNDLE_CONTENT_DIR:selecta>/Frameworks")
  add_custom_command(TARGET selecta_bundle_modules POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${_sel_frameworks_dir}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:selecta_core>" "${_sel_frameworks_dir}/"
    VERBATIM)
  set_property(TARGET selecta APPEND PROPERTY BUILD_RPATH "@executable_path/../Frameworks")
  foreach(_m IN LISTS _sel_bundle_mods)
    add_custom_command(TARGET selecta_bundle_modules POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:${_m}>" "${_sel_plugins_dir}/"
      VERBATIM)
    # I706d: a module that dlopen()s a binary or must ship licence texts keeps them in a folder of its own beside it
    # (sunvox_lib/: the SunVox Library, every licence TXT, CREDITS.txt). The folder travels whole; the About view reads CREDITS.txt.
    get_target_property(_sel_data_dir ${_m} SEL_MODULE_DATA_DIR)
    if(_sel_data_dir)
      get_filename_component(_sel_data_name "${_sel_data_dir}" NAME)
      add_custom_command(TARGET selecta_bundle_modules POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${_sel_data_dir}" "${_sel_plugins_dir}/${_sel_data_name}"
        VERBATIM)
    endif()
  endforeach()
  # I1129: `--target selecta` must refresh the copies too. The copy target depends on the core and every module, and selecta
  # depends on the copy target (not the reverse), so building selecta alone runs the copies; the old reverse edge left an
  # ALL-only target that --target selecta never reached, and ctest ran against a stale Frameworks/libselecta_core.dylib.
  add_dependencies(selecta_bundle_modules selecta_core ${_sel_bundle_mods})
  add_dependencies(selecta selecta_bundle_modules)
endif()

# I1137: the prebuilt empty library database (library_template, src/library/CMakeLists.txt) travels with every build of the app, in the
# directory sel_library_open looks in: Contents/Resources of the macOS bundle and the executable's directory elsewhere (sel_app_data_dir,
# src/library/db/db.c), which on the Amiga is PROGDIR: beside PROGDIR:library.db. `--target selecta` builds and copies it, so a fresh
# start logs `template copied` instead of running the DDL. The Amiga dist (cmake/selecta_dist.cmake) stages it from the same place.
if(TARGET library_template)
  if(APPLE)
    set(_sel_tpl_dir "$<TARGET_BUNDLE_CONTENT_DIR:selecta>/Resources")
  else()
    set(_sel_tpl_dir "$<TARGET_FILE_DIR:selecta>")
  endif()
  add_dependencies(selecta library_template)
  add_custom_command(TARGET selecta POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${_sel_tpl_dir}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_BINARY_DIR}/library-template.db" "${_sel_tpl_dir}/"
    VERBATIM)
endif()

# I1137: the Selecta Amiga archive tree. cmake/selecta_dist.cmake stages Selecta/ (program, library-template.db, Install-Selecta, rexx);
# the CI step packs it. The host test selecta-dist-layout runs the same staging with stand-in files.
if(SEL_AMIGA AND TARGET library_template)
  add_custom_target(selecta_dist
    COMMAND ${CMAKE_COMMAND} -DMODE=stage "-DDEST=${CMAKE_BINARY_DIR}/dist/Selecta" "-DBIN_DIR=$<TARGET_FILE_DIR:selecta>"
      "-DROOT=${PROJECT_SOURCE_DIR}" -P "${PROJECT_SOURCE_DIR}/cmake/selecta_dist.cmake"
    VERBATIM COMMENT "Staging the Selecta Amiga archive tree")
  add_dependencies(selecta_dist selecta library_template)
endif()
